\section{Motivation}
\begin{frame}{Why this project}
\begin{columns}[T]
\begin{column}{0.56\textwidth}
\begin{itemize}
\item \textbf{Ascon} is the NIST lightweight-crypto standard (SP~800-232) for
      authenticated encryption.
\item On a small CPU, computing it in software is slow --- lots of 64-bit work on
      a 32-bit core.
\item \textbf{Goal:} a hardware accelerator that does Ascon-AEAD128, usable from
      C, built with \emph{only open tools}, and \emph{proven on real silicon}.
\end{itemize}
\end{column}
\begin{column}{0.44\textwidth}
\begin{block}{Four objectives}
\small
\begin{enumerate}
\item Hardware Ascon, matching the standard
\item Clean C interface on RISC-V
\item Open, reproducible toolchain
\item Validate on FPGA + measure speed-up
\end{enumerate}
\end{block}
\end{column}
\end{columns}
\end{frame}

\section{Challenges}
\begin{frame}{From ``simulation passes'' to ``correct on silicon''}
\small Eight real bugs --- each diagnosed from concrete evidence, not guesswork:
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\begin{enumerate}
\item Read-only build environment
\item Toolchain name mismatch
\item Float-ABI (double vs.\ soft) mismatch
\item \textbf{Stack past end of RAM} $\to$ silent trap
\end{enumerate}
\end{column}
\begin{column}{0.5\textwidth}
\begin{enumerate}\setcounter{enumi}{4}
\item Flaky USB debugger / shifting ports
\item \alert{\textbf{XBUS bus handshake}}
\item FPGA fit (drop M extension)
\item \alert{\textbf{START-first protocol order}}
\end{enumerate}
\end{column}
\end{columns}
\vspace{2mm}
\begin{block}{The two that mattered}
\small A processor panic with a precise fault address (\code{0x9000\,0000},
\code{0x9000\,0044}) named exactly which register --- and mechanism --- had failed.
\end{block}
\end{frame}

\begin{frame}{The key hardware bug: the bus handshake}
\begin{columns}[T]
\begin{column}{0.54\textwidth}
\small
\textbf{NEORV32's XBUS:}
\begin{itemize}
\item pulses the strobe for \emph{one cycle}
\item samples the acknowledge only in the \emph{next} cycle
\end{itemize}
\vspace{2mm}
\textbf{First adapter:} acknowledged \emph{during} the strobe --- the one cycle the
CPU wasn't looking.
\vspace{2mm}
$\Rightarrow$ CPU never sees ack $\to$ bus timeout $\to$ \alert{access fault}.
\vspace{2mm}
\textbf{Fix:} a handshake FSM that latches the strobe and acks the following cycle;
re-verified against a NEORV32-faithful master.
\end{column}
\begin{column}{0.46\textwidth}
\centering
\begin{tikzpicture}[font=\scriptsize, >=Latex, yscale=0.75]
\draw[->](0,0)--(6.2,0) node[right]{cyc};
\foreach \x in {0,1,...,5}\draw[gray!40](\x,-0.2)--(\x,3.2);
% strobe: pulse in cycle 0
\node[left] at (-0.1,2.7){stb};
\draw[docblue,very thick](0,2.4)--(0,3.0)--(1,3.0)--(1,2.4)--(5,2.4);
% pending
\node[left] at (-0.1,1.5){pending};
\draw[doclblue,very thick](0,1.2)--(1,1.2)--(1,1.8)--(3,1.8)--(3,1.2)--(5,1.2);
% ack (correct): cycle after strobe
\node[left] at (-0.1,0.3){ack (fixed)};
\draw[okgreen,very thick](0,0)--(1,0)--(1,0.6)--(2,0.6)--(2,0)--(5,0);
\node[warnamber, font=\tiny] at (0.5,-0.55){\shortstack{old ack\\here (miss)}};
\draw[warnamber,->,thick](0.5,-0.3)--(0.5,2.6);
\end{tikzpicture}
\end{column}
\end{columns}
\end{frame}

\section{Results}
\begin{frame}{Results: correct, and 18--57$\times$ faster}
\small All 8 cases correct on hardware (\code{enc\_ok}, \code{dec\_ok},
\code{tag\_valid} all set). Cycles at 27\,MHz:
\begin{center}
\renewcommand{\arraystretch}{1.05}
\begin{tabular}{lccrrc}

\begin{frame}{Faster than \emph{what}? --- being honest}
\begin{block}{The comparison}
\small \textbf{Software} = CPU cycles (\code{rdcycle}) for a reference C
Ascon-AEAD128 on the same \code{rv32i} core. \quad
\textbf{Hardware} = the accelerator's busy-cycle counter, same 27\,MHz.
\end{block}
\vspace{1mm}
\alert{Caveats (stated plainly):}
\begin{itemize}\small
\item Software is a \emph{reference}, not hand-optimised --- an optimised version
      would narrow the gap (hardware still wins).
\item The hardware counter excludes register setup + readback; a full end-to-end
      wall-clock number is lower, especially for the tiny \code{empty} case.
\item The $\sim$18$\times$ on real payloads is closer to end-to-end than the
      $\sim$57$\times$ on the empty one.
\end{itemize}
\vspace{1mm}
{\small Fair phrasing: \textbf{the hardware engine vs.\ reference software on the
same SoC} --- same platform, same clock.}
\end{frame}

\section{Wrap-up}
\begin{frame}{Limitations \& future work}
\begin{columns}[T]
\begin{column}{0.5\textwidth}
\textbf{Limitations}
\begin{itemize}\small
\item 32-byte payload cap
\item MMIO word-at-a-time = data-movement bottleneck
\item AEAD128 only on-board
\item No side-channel analysis
\end{itemize}
\end{column}
\begin{column}{0.5\textwidth}
\textbf{Future work}
\begin{itemize}\small
\item Rate-block streaming (arbitrary length)
\item DMA / wider datapath
\item Hash \& XOF modes on-chip
\item End-to-end wall-clock metric
\end{itemize}
\end{column}
\end{columns}
\end{frame}

\begin{frame}{Conclusion}
\begin{block}{What was achieved}
\small A standard-conformant Ascon-AEAD128 accelerator --- generated from a
verified model, usable from C on a RISC-V SoC, built with \emph{only open tools},
and \textbf{validated on silicon}: all 8 cases correct, \textbf{18--57$\times$}
faster than reference software on the same core.
\end{block}
\vspace{2mm}
\begin{itemize}
\item The hard part was the \textbf{hardware/software boundary} --- the bus
      handshake and the protocol order --- exactly what only real, end-to-end
      validation exposes.
\item Every bug was found from the board's own evidence.
\end{itemize}
